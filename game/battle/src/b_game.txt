   色は盤の色表（PAL）に入っている値だけを使う。表に無い色は丸められて、

/* 同じ属性の中で、上から順に色の帯を配る。札を足すときは el を書くだけでよい */
(function(){
  const seen = {};
  DECK.forEach(c => {
    const e = ELEM[c.el] || ELEM.norm;
    const k = seen[c.el] = (seen[c.el] === undefined ? 0 : seen[c.el] + 1);
    c.col = e.cols[k % e.cols.length];
  });
})();

/** 区画の並び。同じ札を固めると、機械投擲（初期条件が近い）では
    隣どうしに固まって出る。残りがいちばん多い種類から、直前と違うものを置く。 */
function layout(counts){
  const left = counts.slice(), out = [];
  while (left.some(v => v > 0)){
    let pick = -1;
    for (let i=0;i<left.length;i++){
      if (left[i] <= 0) continue;
      if (i === out[out.length-1] && left.filter(v=>v>0).length > 1) continue;
      if (pick < 0 || left[i] > left[pick]) pick = i;
    }
    out.push(pick); left[pick]--;
  }
  return out;
}

/** ダメージ。盾から先に削る。
    brk … 盾を割ってから当てる（ハンマー）
    raw … 盾を無視して直接 HP を削る（火傷） */
function hit(t, n, brk, raw){
  if (brk) t.shield = 0;
  if (raw){ t.hp = Math.max(0, t.hp - n); return; }
  const a = Math.min(t.shield, n);
  t.shield -= a; t.hp = Math.max(0, t.hp - (n - a));
}
