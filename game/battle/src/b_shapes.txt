function glyph(k, px){
  const r = GLYPH[k].map(([x,y,w,h,rr,ss])=>
    `<rect x="${x}" y="${y}" width="${w}" height="${h}" fill="${C(rr,ss)}"/>`).join("");
  return `<svg width="${px}" height="${px}" viewBox="0 0 8 8" shape-rendering="crispEdges">${r}</svg>`;
}
